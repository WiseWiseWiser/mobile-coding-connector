import (
	"os"
	"strings"
	"testing"

	"github.com/xhd2015/doctest/session"
)

func Assert(t *testing.T, _ *session.Doctest, req *Request, resp *Response, err error) {
	if err != nil {
		t.Fatalf("Run error: %v", err)
	}
	if resp.ExitCode != 0 {
		t.Fatalf("exit %d; combined:\n%s", resp.ExitCode, resp.Combined)
	}
	if !strings.Contains(resp.Stdout, "removed alias "+aliasLeafName) {
		t.Fatalf("stdout missing the removal line:\n%s", resp.Stdout)
	}
	if !strings.Contains(resp.Stderr, "warning: not removing ~/.local/bin/"+aliasLeafBinary) {
		t.Fatalf("stderr missing the foreign-file warning:\n%s", resp.Stderr)
	}
	if _, statErr := os.Stat(resp.Wrapper); statErr != nil {
		t.Fatalf("user script was removed: %v", statErr)
	}
	if resp.WrapperAfter != aliasForeignBody {
		t.Fatalf("user script changed: %q", resp.WrapperAfter)
	}
	if got := aliasStoreNames(t, resp.StoreJSON); len(got) != 0 {
		t.Fatalf("store still has aliases: %v\n%s", got, resp.StoreJSON)
	}
}
