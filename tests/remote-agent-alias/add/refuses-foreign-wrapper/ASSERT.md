import (
	"strings"
	"testing"

	"github.com/xhd2015/doctest/session"
)

func Assert(t *testing.T, _ *session.Doctest, req *Request, resp *Response, err error) {
	if err != nil {
		t.Fatalf("Run error: %v", err)
	}
	if resp.ExitCode != 1 {
		t.Fatalf("exit %d, want 1; combined:\n%s", resp.ExitCode, resp.Combined)
	}
	if !strings.Contains(resp.Stderr, "Error:") {
		t.Fatalf("stderr missing the Error prefix:\n%s", resp.Stderr)
	}
	if !strings.Contains(resp.Stderr, "was not generated by remote-agent alias; pass --force") {
		t.Fatalf("stderr missing the --force hint:\n%s", resp.Stderr)
	}
	if resp.WrapperAfter != aliasForeignBody {
		t.Fatalf("user script was modified: %q", resp.WrapperAfter)
	}
	if resp.StoreJSON != "" {
		t.Fatalf("failed add must not write the store: %s", resp.StoreJSON)
	}
}
